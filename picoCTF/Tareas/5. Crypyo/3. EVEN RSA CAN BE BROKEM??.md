# N = p * q
# Según el comentario de tu imagen, p es 2
p = 2
q = N // p

# Calcular phi(N) = (p - 1) * (q - 1)
phi = (p - 1) * (q - 1)

# Calcular clave privada d
d = inverse(e, phi)

# Descifrar el mensaje
# m = c^d mod N
m = pow(c, d, N)

# Convertir de número a bytes
flag = long_to_bytes(m)

# Imprimir el resultado, ignorando errores de codificación
print(flag.decode(errors="ignore"))
